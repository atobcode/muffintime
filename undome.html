#!/usr/bin/env python3
"""
main.py - IPCP 418 nightly monitor

Behavior:
 - Runs only during the test window START_HOUR -> STOP_HOUR (defaults 22 -> 6).
 - Enforces a rolling MAX_HITS per WINDOW_HOURS (defaults 420 per 12h).
 - Sends a one-time "no-show" alert after ALERT_AFTER seconds if 418 has not been seen.
 - On seeing HTTP 418, sends an alert (email and optional SMS-via-email), saves last_response.html and exits.
 - Exits automatically at STOP_HOUR each night.

Environment variables (examples / defaults):
 - TARGET_URL="https://fox-towing.com/contact-us"
 - START_HOUR=22
 - STOP_HOUR=6
 - ALERT_AFTER=180
 - POLL_INTERVAL=60
 - TIMEOUT=10
 - MAX_HITS=420
 - WINDOW_HOURS=12
 - HITS_FILE=/var/lib/ipcp-monitor/hits.log
 - TO_EMAIL=foxtowing2021@gmail.com
 - SMS_TO=
 - SMTP_HOST=smtp.example.com
 - SMTP_PORT=587
 - SMTP_USER=
 - SMTP_PASS=
 - SMTP_FROM=monitor@localhost
"""
import os
import time
import urllib.request
import urllib.error
import smtplib
from email.message import EmailMessage
from datetime import datetime
from collections import deque
import sys

# Configuration from environment
TARGET_URL = os.getenv("TARGET_URL", "https://fox-towing.com/contact-us")
START_HOUR = int(os.getenv("START_HOUR", "22"))
STOP_HOUR = int(os.getenv("STOP_HOUR", "6"))
ALERT_AFTER = int(os.getenv("ALERT_AFTER", "180"))  # seconds; 0 disables no-show alert
POLL_INTERVAL = float(os.getenv("POLL_INTERVAL", "60"))
TIMEOUT = float(os.getenv("TIMEOUT", "10"))

MAX_HITS = int(os.getenv("MAX_HITS", "420"))
WINDOW_HOURS = float(os.getenv("WINDOW_HOURS", "12"))
WINDOW_SEC = WINDOW_HOURS * 3600.0
HITS_FILE = os.getenv("HITS_FILE", "hits.log")

TO_EMAIL = os.getenv("TO_EMAIL", "foxtowing2021@gmail.com")
SMS_TO = os.getenv("SMS_TO", "")  # optional: an email-to-SMS address (carrier gateway)
SMTP_HOST = os.getenv("SMTP_HOST", "smtp.example.com")
SMTP_PORT = int(os.getenv("SMTP_PORT", "587"))
SMTP_USER = os.getenv("SMTP_USER")
SMTP_PASS = os.getenv("SMTP_PASS")
SMTP_FROM = os.getenv("SMTP_FROM", SMTP_USER or "monitor@localhost")

# Helpers: time window checks
def now_hour():
    return datetime.utcnow().hour  # use UTC for scheduling consistency

def in_test_window(start_hour: int, stop_hour: int) -> bool:
    h = now_hour()
    if start_hour == stop_hour:
        return True  # all day
    if start_hour < stop_hour:
        return start_hour <= h < stop_hour
    # overnight window (e.g., 22 -> 6)
    return h >= start_hour or h < stop_hour

# Persistent hit storage
def load_hits(path: str, window_sec: float):
    now = time.time()
    hits = deque()
    if not os.path.exists(path):
        return hits
    try:
        with open(path, "r", encoding="utf-8") as fh:
            for ln in fh:
                ln = ln.strip()
                if not ln:
                    continue
                try:
                    ts = float(ln)
                except Exception:
                    continue
                if ts >= now - window_sec:
                    hits.append(ts)
    except Exception:
        return deque()
    return hits

def persist_hit(path: str, ts: float):
    try:
        d = os.path.dirname(path)
        if d and not os.path.exists(d):
            os.makedirs(d, exist_ok=True)
        with open(path, "a", encoding="utf-8") as fh:
            fh.write(f"{ts}\n")
    except Exception:
        pass

def prune_hits(hits: deque, window_sec: float):
    now = time.time()
    cutoff = now - window_sec
    while hits and hits[0] < cutoff:
        hits.popleft()

def ensure_rate_allowance(hits: deque, max_hits: int, window_sec: float):
    while True:
        prune_hits(hits, window_sec)
        if len(hits) < max_hits:
            return
        earliest = hits[0]
        next_allowed = earliest + window_sec
        now = time.time()
        sleep_for = max(1.0, next_allowed - now)
        eta = datetime.utcfromtimestamp(now + sleep_for).isoformat() + "Z"
        print(f"[RATE LIMIT] {len(hits)} hits in window >= {max_hits}. Sleeping {int(sleep_for)}s until {eta}")
        time.sleep(sleep_for)

# HTTP fetch
def fetch(url):
    req = urllib.request.Request(url, headers={"User-Agent": "ipcp-monitor/1.0"})
    try:
        with urllib.request.urlopen(req, timeout=TIMEOUT) as r:
            return r.getcode(), r.read().decode(errors="replace")
    except urllib.error.HTTPError as e:
        try:
            body = e.read().decode(errors="replace")
        except Exception:
            body = ""
        return e.code, body
    except Exception:
        return None, ""

# Email send (STARTTLS + LOGIN)
def send_via_smtp(host, port, user, passwd, mail_from, mail_to, subject, body):
    msg = EmailMessage()
    msg["From"] = mail_from
    msg["To"] = mail_to
    msg["Subject"] = subject
    msg.set_content(body)
    try:
        with smtplib.SMTP(host, port, timeout=15) as s:
            s.ehlo()
            s.starttls()
            s.ehlo()
            if user and passwd:
                s.login(user, passwd)
            s.send_message(msg)
        return True
    except Exception as e:
        print("[EMAIL ERROR]", e)
        return False

def notify(subject: str, body: str):
    # Send to main email
    ok = send_via_smtp(SMTP_HOST, SMTP_PORT, SMTP_USER, SMTP_PASS, SMTP_FROM, TO_EMAIL, subject, body)
    if SMS_TO:
        # send same body to SMS gateway address (best-effort)
        send_via_smtp(SMTP_HOST, SMTP_PORT, SMTP_USER, SMTP_PASS, SMTP_FROM, SMS_TO, subject, body)
    return ok

def save_html(text: str):
    try:
        with open("last_response.html", "w", encoding="utf-8") as fh:
            fh.write(text or "")
    except Exception:
        pass

def should_stop_by_time(start_hour: int, stop_hour: int) -> bool:
    # Called periodically to decide if we reached stop time and should exit
    return not in_test_window(start_hour, stop_hour)

def iso_now():
    return datetime.utcnow().isoformat() + "Z"

def main():
    # If invoked outside the window, exit immediately
    if not in_test_window(START_HOUR, STOP_HOUR):
        print(f"[INFO] Current UTC hour {now_hour()} is outside the test window {START_HOUR}->{STOP_HOUR}. Exiting.")
        return 0

    hits = load_hits(HITS_FILE, WINDOW_SEC)
    start_ts = time.time()
    alerted_no418 = False

    print(f"[INFO] Starting monitor for {TARGET_URL} between {START_HOUR}:00 and {STOP_HOUR}:00 UTC")
    print(f"[INFO] Rate limit: {MAX_HITS} hits per {WINDOW_HOURS} hours. ALERT_AFTER={ALERT_AFTER}s")

    while True:
        # If stop time reached while running, exit gracefully
        if should_stop_by_time(START_HOUR, STOP_HOUR):
            print(f"[INFO] Reached stop time (UTC hour {now_hour()}). Exiting monitor.")
            return 0

        # Rate-limit enforcement
        ensure_rate_allowance(hits, MAX_HITS, WINDOW_SEC)

        # Record hit immediately
        ts = time.time()
        hits.append(ts)
        persist_hit(HITS_FILE, ts)

        code, body = fetch(TARGET_URL)
        now = iso_now()
        print(f"[{now}] status={code} (hits_in_window={len(hits)})")

        # If 418 observed: notify, save HTML, self-kill (exit)
        if code == 418:
            snippet = (body or "").replace("\\n", " ")[:1000]
            subject = f"[ALERT] {TARGET_URL} returned 418"
            body_txt = f"Target: {TARGET_URL}\\nStatus: 418\\nTime (UTC): {now}\\n\\nSnippet:\\n{snippet}\\n\\nMonitor will now exit for pro intervention."
            notify(subject, body_txt)
            save_html(body or "")
            print("[SENT] 418 alert sent. Exiting (self-kill) for pro handoff.")
            return 0

        # If no-418 alert configured and time elapsed, send one-time "no-show" notification
        if ALERT_AFTER > 0 and not alerted_no418:
            elapsed = time.time() - start_ts
            if elapsed >= ALERT_AFTER:
                snippet = (body or "").replace("\\n", " ")[:1000]
                subject = f"[ALERT] {TARGET_URL} - no 418 observed after {ALERT_AFTER}s"
                body_txt = f"Target: {TARGET_URL}\\nTime (UTC): {now}\\nElapsed without 418: {int(elapsed)}s\\n\\nLast seen status: {code}\\n\\nLast snippet:\\n{snippet}"
                notify(subject, body_txt)
                print("[SENT] no-show alert sent.")
                alerted_no418 = True

        # Sleep until next poll, but wake up earlier if the stop time hits
        sleep_total = POLL_INTERVAL
        slept = 0.0
        while slept < sleep_total:
            if should_stop_by_time(START_HOUR, STOP_HOUR):
                print(f"[INFO] stop time reached during sleep. Exiting.")
                return 0
            to_sleep = min(5.0, sleep_total - slept)
            time.sleep(to_sleep)
            slept += to_sleep

if __name__ == "__main__":
    try:
        rc = main() or 0
        sys.exit(rc)
    except Exception as exc:
        print("[FATAL ERROR]", exc)
        sys.exit(2)

#!/usr/bin/env python3
"""
control_server.py - minimal control API for starting/stopping monitor.
Usage:
  CONTROL_TOKEN=your-secret-token python3 control_server.py
Endpoints (POST):
  /start  -> starts monitor (runs main_smtp.py)
  /stop   -> stops monitor
  /status -> returns running/stopped
Note: run behind TLS / firewall and use a strong CONTROL_TOKEN.
"""
import os, subprocess, shlex, signal, time, json
from http.server import BaseHTTPRequestHandler, HTTPServer

TOKEN = os.getenv("CONTROL_TOKEN")
MONITOR_CMD = os.getenv("MONITOR_CMD", "python3 main_smtp.py")
PORT = int(os.getenv("CONTROL_PORT", "8001"))

proc = None

class Handler(BaseHTTPRequestHandler):
    def _auth_fail(self):
        self.send_response(403); self.send_header("Content-Type","application/json"); self.end_headers()
        self.wfile.write(b'{"error":"forbidden"}')

    def do_POST(self):
        global proc
        token = self.headers.get("X-Auth-Token")
        if not TOKEN or token != TOKEN:
            return self._auth_fail()
        path = self.path
        if path == "/start":
            if proc and proc.poll() is None:
                self._write({"status":"running","pid":proc.pid})
                return
            proc = subprocess.Popen(shlex.split(MONITOR_CMD), stdout=open("monitor.log","a"), stderr=subprocess.STDOUT)
            time.sleep(0.2)
            self._write({"status":"started","pid":proc.pid})
            return
        if path == "/stop":
            if not proc or proc.poll() is not None:
                self._write({"status":"stopped"})
                return
            proc.terminate()
            try:
                proc.wait(timeout=5)
            except Exception:
                proc.kill()
            self._write({"status":"stopped"})
            return
        if path == "/status":
            running = bool(proc and proc.poll() is None)
            self._write({"status":"running" if running else "stopped", "pid": proc.pid if running else None})
            return
        self.send_response(404); self.end_headers()

    def _write(self, obj):
        data = json.dumps(obj).encode("utf-8")
        self.send_response(200); self.send_header("Content-Type","application/json"); self.send_header("Content-Length",str(len(data))); self.end_headers()
        self.wfile.write(data)

if __name__ == "__main__":
    if not TOKEN:
        print("Set CONTROL_TOKEN env var to a strong secret"); raise SystemExit(1)
    server = HTTPServer(("127.0.0.1", PORT), Handler)
    print(f"Control server listening on 127.0.0.1:{PORT}")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        server.server_close()

function app_base_url() {
    $u = getenv('APP_URL');

        $u = $scheme . '://' . $_SERVER['HTTP_HOST'];

    if ($provider === 'google') {
        $post = [
            'code' => $code,
            'client_id' => $cfg['client_id'],
            'client_secret' => $cfg['client_secret'],
            'redirect_uri' => $cfg['redirect_uri'],
            'grant_type' => 'authorization_code',
        ];
        $resp = http_post_form($cfg['token_url'], $post, ['Accept: application/json']);
        $data = json_decode($resp, true);
        $token = $data['access_token'] ?? null;
    } elseif ($provider === 'github') {
        $post = [
            'client_id' => $cfg['client_id'],
            'client_secret' => $cfg['client_secret'],
            'code' => $code,
            'redirect_uri' => $cfg['redirect_uri'],
            'state' => $_SESSION['oauth2_state'],
        ];
        $resp = http_post_form($cfg['token_url'], $post, ['Accept: application/json']);
        $data = json_decode($resp, true);
        $token = $data['access_token'] ?? null;
    } elseif ($provider === 'twitch') {
        $post = [
            'client_id' => $cfg['client_id'],
            'client_secret' => $cfg['client_secret'],
            'code' => $code,
            'grant_type' => 'authorization_code',
            'redirect_uri' => $cfg['redirect_uri'],
        ];
        $resp = http_post_form($cfg['token_url'], $post, ['Accept: application/json']);
        $data = json_decode($resp, true);
        $token = $data['access_token'] ?? null;
    } else {
        throw new Exception("unsupported provider");
    }
} catch (Exception $e) {
    echo "Token exchange failed: " . htmlspecialchars($e->getMessage());
    exit;

    echo "No access token returned.";
    exit;

function http_post_form($url, $post_fields, $headers = []) {
    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, http_build_query($post_fields));
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    curl_setopt($ch, CURLOPT_HTTPHEADER, array_merge($headers, ['Content-Type: application/x-www-form-urlencoded']));
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, true);
    $resp = curl_exec($ch);
    if ($resp === false) {
        $e = curl_error($ch);
        curl_close($ch);
        throw new Exception("curl error: $e");
    }
    $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    if ($code >= 400) {
        throw new Exception("HTTP $code during token exchange: $resp");
    }
    return $resp;
}

function http_get($url, $headers = []) {
    $ch = curl_init($url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
    if (!empty($headers)) {
        curl_setopt($ch, CURLOPT_HTTPHEADER, $headers);
    }
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, true);
    $resp = curl_exec($ch);
    if ($resp === false) {
        $e = curl_error($ch);
        curl_close($ch);
        throw new Exception("curl error: $e");
    }
    $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    if ($code >= 400) {
        throw new Exception("HTTP $code during userinfo fetch: $resp");
    }
    return $resp;
}

name: IPCP 418 poll
on:
  schedule:
    - cron: '*/5 * * * *'
  workflow_dispatch:
jobs:
  poll:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          persist-credentials: true

      - name: Check test window (exit if outside)
        id: window
        run: |
          START_HOUR=${{ secrets.START_HOUR || '22' }}
          STOP_HOUR=${{ secrets.STOP_HOUR || '6' }}
          HOUR=$(date -u +%H)
          echo "UTC hour: $HOUR, window: ${START_HOUR}->${STOP_HOUR}"
          sh -c '
            sh_hour=$HOUR
            start=$START_HOUR
            stop=$STOP_HOUR
            if [ "$start" = "$stop" ]; then
              exit 0
            fi
            if [ $start -lt $stop ]; then
              if [ $sh_hour -ge $start -a $sh_hour -lt $stop ]; then exit 0; else exit 78; fi
            else
              if [ $sh_hour -ge $start -o $sh_hour -lt $stop ]; then exit 0; else exit 78; fi
            fi
          '

      - name: Do single HTTP GET and capture status
        id: http
        run: |
          TARGET_URL="${{ secrets.TARGET_URL }}"
          echo "GET $TARGET_URL"
          set -o pipefail
          STATUS=$(curl -s -o /tmp/resp_body.txt -w "%{http_code}" "$TARGET_URL" || echo "000")
          echo "status=$STATUS" >> $GITHUB_OUTPUT
          echo "body=$(head -c 200 /tmp/resp_body.txt | sed 's/\"/\\\"/g')" >> $GITHUB_OUTPUT

      - name: If 418 -> send email and create marker
        if: steps.http.outputs.status == '418'
        env:
          SMTP_HOST: ${{ secrets.SMTP_HOST }}
          SMTP_PORT: ${{ secrets.SMTP_PORT }}
          SMTP_USER: ${{ secrets.SMTP_USER }}
          SMTP_PASS: ${{ secrets.SMTP_PASS }}
          EMAIL_FROM: ${{ secrets.EMAIL_FROM }}
          EMAIL_TO: ${{ secrets.EMAIL_TO }}
        uses: dawidd6/action-send-mail@v3
        with:
          server_address: ${{ env.SMTP_HOST }}
          server_port: ${{ env.SMTP_PORT }}
          username: ${{ env.SMTP_USER }}
          password: ${{ env.SMTP_PASS }}
          subject: "ALERT: ${{ secrets.TARGET_URL }} returned 418"
          body: |
            The monitor run at $(date -u) saw HTTP 418 for ${{ secrets.TARGET_URL }}.
            Snippet:
            ${{ steps.http.outputs.body }}
          from: ${{ env.EMAIL_FROM }}
          to: ${{ env.EMAIL_TO }}

      - name: Create marker file (on 418)
        if: steps.http.outputs.status == '418'
        run: |
          TODAY=$(date -u +%F)
          MARKER="monitor-markers/418-$TODAY.txt"
          echo "418 detected at $(date -u)" > "$MARKER"
          git add "$MARKER"
          git commit -m "marker: 418 detected $TODAY" || echo "no changes to commit"
          git push
